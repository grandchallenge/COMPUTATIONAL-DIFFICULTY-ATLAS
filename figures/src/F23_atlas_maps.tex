\head{1. Class containment}{Solid inclusion symbols denote proved set inclusions; equality may be open.}
\boxline{\(\C P\subseteq\C{NP}\subseteq\C{PH}\subseteq\C{PSPACE}\subseteq\C{EXPTIME}\)}
\boxline{\(\C P\subseteq\C{BQP}\subseteq\C{PSPACE}\)}
\boxline{\(\C P\subseteq\C{coNP}\subseteq\C{PSPACE}\)}
\textbf{Strict endpoint:} \(\C P\ne\C{EXPTIME}\) (time hierarchy).\\
No adjacent strictness follows from this endpoint alone.\\
No general inclusion between \(\C{NP}\) and \(\C{BQP}\) is asserted.

\head{2. Reduction relations}{Arrows translate source tasks to target tasks; the labels determine what transfers.}
\boxline{\( A\xrightarrow{\ f\ } B,\quad x\in A\ \Longleftrightarrow\ f(x)\in B\)\\
A solver for the target solves the source after translation.}
\begin{tabular}{@{}p{120pt}p{280pt}@{}}
\textbf{Edge type} & \textbf{Additional contract}\\ \hline
Polynomial many-one & Polynomial translation time\\
FPT many-one & Parameter-bounded output and FPT time\\
Gap-preserving & Explicit separation of objective values\\
Distributional & Control of pushforward probability\\
Computable many-one & Total computable translation\\
Turing & Adaptive oracle queries; may negate answers
\end{tabular}

\head{4. Proved separations}{Strictness is stated only where supported by a theorem.}
\boxline{\(\C P\subsetneq\C{EXPTIME}\)\\Deterministic time hierarchy}
\boxline{\(\mathbf 0<_T\mathbf 0'<_T\mathbf 0''<_T\cdots\)\\
Strict Turing jumps; oracle computability, not running time}
\boxline{A large proof-width requirement in a named proof system\\
does not imply a lower bound on all decision algorithms.}
These are different theorems about different relations.

\begin{tabular}{@{}p{120pt}p{125pt}p{145pt}@{}}
& Uniform bits & All-zero point mass\\ \hline
